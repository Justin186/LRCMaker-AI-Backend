    function applyLrcToEditor() {
        const targetContent = currentMode === 'standard' ? cachedStandardLrc : cachedEnhancedLrc;
        if (!targetContent) return;
        if (typeof GM_setClipboard !== 'undefined') GM_setClipboard(targetContent, 'text');
        else navigator.clipboard.writeText(targetContent);

        const targetTextarea = document.querySelector('textarea:not(#ai-lyrics)');
        if (targetTextarea) {
            targetTextarea.value = targetContent;
            targetTextarea.dispatchEvent(new Event('input', { bubbles: true }));
            targetTextarea.dispatchEvent(new Event('change', { bubbles: true }));
        }
    }

    function parseEnhancedLrc(lrc) {
        const lines = [];
        // ESLRC 风格：行首 [mm:ss.xx] 为行起始时间
        // 每个词后跟 [mm:ss.xx] 为该词结束时间；若词间有停顿，则额外跟一个 [mm:ss.xx] 为下一词开始时间
        // 例（无停顿）：[00:12.34]我[00:13.00]爱[00:13.50]你[00:14.20]
        // 例（词2后有停顿）：[00:12.34]我[00:13.00]爱[00:13.50][00:15.00]你[00:15.80]
        const lineStartReg = /^\[(\d{2}):(\d{2}\.\d{2})\]/;
        const wordReg = /([^\[\]]+)((?:\[\d{2}:\d{2}\.\d{2}\])+)/g;
        const timeReg = /\[(\d{2}):(\d{2}\.\d{2})\]/g;

        lrc.split('\n').forEach(lineStr => {
            const line = { words: [] };
            const lineStartMatch = lineStartReg.exec(lineStr);
            let nextStart = lineStartMatch
                ? parseInt(lineStartMatch[1]) * 60 + parseFloat(lineStartMatch[2])
                : null;

            let match;
            wordReg.lastIndex = 0;
            while ((match = wordReg.exec(lineStr)) !== null) {
                const text = match[1].trim();
                if (!text) continue;

                const stamps = [];
                let tm;
                timeReg.lastIndex = 0;
                while ((tm = timeReg.exec(match[2])) !== null) {
                    stamps.push(parseInt(tm[1]) * 60 + parseFloat(tm[2]));
                }

                const end = stamps[0];
                const start = nextStart !== null ? nextStart : end;
                line.words.push({ start: start, text: text, end: end });

                // 有第二个时间戳说明词间有停顿，下一词从该时间戳开始；否则无缝衔接
                nextStart = stamps.length > 1 ? stamps[1] : end;
            }
            if (line.words.length > 0) lines.push(line);
        });
        return lines;
    }

    function buildKtvStage() {
        ktvDisplay.innerHTML = '';
        parsedKtvData = parseEnhancedLrc(cachedEnhancedLrc);

        parsedKtvData.forEach((line, lineIdx) => {
            const lineDiv = document.createElement('div');
            lineDiv.className = 'ktv-line';
            lineDiv.id = `ktv-line-${lineIdx}`;

            line.words.forEach(word => {
                const span = document.createElement('span');
                span.className = 'ktv-word';
                span.setAttribute('data-text', word.text);
                span.setAttribute('data-start', word.start);
                span.setAttribute('data-end', word.end);
                span.innerText = word.text;
                lineDiv.appendChild(span);
            });
            ktvDisplay.appendChild(lineDiv);
        });
    }

    let animationFrameId = null;
    let activeLineIndex = -1;

    function renderLoop() {
        if (ktvAudio.paused) return;

        const currentTime = ktvAudio.currentTime;
        const words = document.querySelectorAll('.ktv-word');

        words.forEach(word => {
            const start = parseFloat(word.getAttribute('data-start'));
            const end = parseFloat(word.getAttribute('data-end'));

            if (currentTime < start) {
                word.style.setProperty('--progress', '0%');
            } else if (currentTime > end) {
                word.style.setProperty('--progress', '100%');
            } else {
                const percent = ((currentTime - start) / (end - start)) * 100;
                word.style.setProperty('--progress', `${percent}%`);

                const parentLine = word.parentElement;
                const lineId = parentLine.id;
                const lineIdx = parseInt(lineId.split('-')[2]);

                if (activeLineIndex !== lineIdx) {
                    activeLineIndex = lineIdx;
                    document.querySelectorAll('.ktv-line').forEach(l => l.classList.remove('active'));
                    parentLine.classList.add('active');
                    parentLine.scrollIntoView({ behavior: "smooth", block: "center" });
                }
            }
        });

        animationFrameId = requestAnimationFrame(renderLoop);
    }

    ktvAudio.addEventListener('play', () => {
        animationFrameId = requestAnimationFrame(renderLoop);
    });

    ktvAudio.addEventListener('seeked', () => {
        renderLoop();
    });

    async function findActiveBackendPort(startPort = 8000, maxTries = 10) {
        for (let port = startPort; port < startPort + maxTries; port++) {
            try {
                const controller = new AbortController();
                const timeoutId = setTimeout(() => controller.abort(), 300);

                const response = await fetch(`http://127.0.0.1:${port}/api/ping`, {
                    signal: controller.signal
                });
                clearTimeout(timeoutId);

                if (response.ok) {
                    const data = await response.json();
                    if (data.app === "lrc-maker-ai") {
                        console.log(`✅ 成功探测到 AI 后端运行在端口: ${port}`);
                        return port;
                    }
                }
            } catch (e) {
                console.log(`端口 ${port} 未响应，尝试下一个...`);
            }
        }
        throw new Error("未探测到运行中的本地 AI 引擎，请确认黑框程序已启动！");
    }

    btnAlign.addEventListener('click', async () => {
        let audioFile = null;

        // 🌟 v2.1 终极绝招：直接从 LRC-Maker 网页的播放器中拦截 Blob 数据 (无视用户的载入方式)
        const pageAudio = document.querySelector('audio:not(#ktv-audio)'); 
        if (pageAudio && pageAudio.src && pageAudio.src.startsWith('blob:')) {
            try {
                btnAlign.disabled = true;
                btnAlign.style.background = '#9e9e9e';
                btnAlign.innerText = '📡 正在准备音频数据...';
                
                const response = await fetch(pageAudio.src);
                const blob = await response.blob();
                audioFile = new File([blob], "audio_intercepted.wav", { type: blob.type }); 
                console.log("✅ 成功从网页播放器中提取到音频 Blob!");
            } catch (e) {
                console.log("⚠️ 从 Blob 提取音频失败，准备回退查找", e);
            }
        }

        // 🛡️ 备用方案：如果上面失败了，再回退到传统的 input 标签查找
        if (!audioFile) {
            const fileInputs = document.querySelectorAll('input[type="file"]');
            for (let input of fileInputs) {
                if (input.files && input.files.length > 0) {
                    const file = input.files[0];
                    if (file.type.startsWith('audio/') || file.name.endsWith('.mp3') || file.name.endsWith('.wav')) {
                        audioFile = file;
                        break;
                    }
                }
            }
        }

        if (!audioFile) {
            alert('❌ 找不到音频文件！请确认是否已在网页左下角载入音频（或直接拖拽）。');
            btnAlign.disabled = false;
            btnAlign.style.background = '#2196F3';
            btnAlign.innerText = '🚀 一键 AI 强制对齐';
            return;
        }

        const lyricsText = lyricsInput.value.trim();
        if (!lyricsText) {
            alert('❌ 请在文本框中粘贴歌词！');
            btnAlign.disabled = false;
            btnAlign.style.background = '#2196F3';
            btnAlign.innerText = '🚀 一键 AI 强制对齐';
            return;
        }

        const formData = new FormData();
        formData.append('audio', audioFile);
        formData.append('lyrics', lyricsText);
        formData.append('ti', document.getElementById('ai-ti').value.trim());
        formData.append('ar', document.getElementById('ai-ar').value.trim());
        formData.append('al', document.getElementById('ai-al').value.trim());

        try {
            btnAlign.disabled = true;
            btnAlign.style.background = '#9e9e9e';

            btnAlign.innerText = '📡 正在寻找本地 AI 引擎...';
            statusTxt.innerText = '正在扫描端口...';
            const activePort = await findActiveBackendPort();

            btnAlign.innerText = '⏳ AI 正在听歌... (约需几十秒)';
            statusTxt.innerText = `已连接至端口 ${activePort}`;

            if(audioBlobUrl) URL.revokeObjectURL(audioBlobUrl);
            audioBlobUrl = URL.createObjectURL(audioFile);
            ktvAudio.src = audioBlobUrl;

            const response = await fetch(`http://127.0.0.1:${activePort}/api/align`, { method: 'POST', body: formData });
            const resJson = await response.json();

            if (resJson.code === 200) {
                const lrcData = resJson.data;
                cachedStandardLrc = typeof lrcData === 'string' ? lrcData : lrcData.standard_lrc;
                cachedEnhancedLrc = typeof lrcData === 'string' ? lrcData : lrcData.enhanced_lrc;

                applyLrcToEditor();
                buildKtvStage();

                if (currentMode === 'enhanced') {
                    previewContainer.style.display = 'block';
                }

                setTimeout(() => {
                    btnAlign.style.background = '#2196F3';
                    btnAlign.innerText = '🚀 再次强制对齐';
                    statusTxt.innerText = '生成完毕！可点击顶部 Tab 预览效果';
                    btnAlign.disabled = false;
                }, 2000);
            } else {
                throw new Error(resJson.message);
            }
        } catch (err) {
            btnAlign.style.background = '#f44336';
            btnAlign.innerText = '❌ 生成失败';
            statusTxt.innerText = err.message;
            setTimeout(() => {
                btnAlign.style.background = '#2196F3';
                btnAlign.innerText = '🚀 重新尝试';
                btnAlign.disabled = false;
            }, 3000);
        }
    });
})();
